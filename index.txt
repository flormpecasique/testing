        function initTheme() {
            const btn = document.getElementById('theme-toggle');

            if (savedTheme === 'dark' || (!savedTheme && window.matchMedia('(prefers-color-scheme: dark)').matches)) {

        // --- LÓGICA DE ESCALADO DE PREVIEW ---
        function handleResize() {
            const container = document.getElementById('preview-scaler').parentElement;
            const scaler = document.getElementById('preview-scaler');
            const containerWidth = container.clientWidth - 32; // padding
            const a4Width = 794; // Aprox 210mm en px (96dpi)
            
            if (containerWidth < a4Width) {
                const scale = containerWidth / a4Width;
                scaler.style.transform = `scale(${scale})`;
                scaler.style.height = `${1123 * scale}px`; // 297mm approx
            } else {
                scaler.style.transform = `scale(1)`;
                scaler.style.height = 'auto';
            }
        }

        // --- GESTIÓN DE CÓDIGO QR ---
        function handleQRUpload(event) {
            const file = event.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = (e) => {
                const img = new Image();
                img.onload = () => {
                    // Guardar imagen original para el PDF
                    state.qrImageSrc = e.target.result;

                    // Decodificar con jsQR
                    const canvas = document.getElementById('qr-canvas');
                    const ctx = canvas.getContext('2d', { willReadFrequently: true });
                    canvas.width = img.width;
                    canvas.height = img.height;
                    ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
                    
                    const imageData = ctx.getImageData(0, 0, canvas.width, canvas.height);
                    const code = jsQR(imageData.data, imageData.width, imageData.height, {
                        inversionAttempts: "dontInvert",
                    });

                    if (code) {
                        state.qrRawText = code.data;
                        parseQRData(code.data);
                        showToast('QR decodificado y datos extraídos correctamente.', 'success');
                    } else {
                        // Si no es un QR legible pero es imagen, la usamos y avisamos
                        showToast('Imagen cargada, pero no se detectó texto QR. Se añadirá visualmente.', 'info');
                        updatePreview();
                    }
                };
                img.src = e.target.result;
            };
            reader.readAsDataURL(file);
        }

        function parseQRData(text) {
            // Simulador de extracción de datos de la URL de cotejo (VeriFactu)
            // Normalmente en España llevarían params tipo ?IDEmisorFactura=...
            
            try {
                // Buscamos patrones comunes. Si es una URL:
                if (text.includes('?')) {
                    const urlParams = new URLSearchParams(text.split('?')[1]);
                    
                    // Mapeo hipotético SIF
                    if (urlParams.get('IDEmisorFactura')) state.data.nifEmisor = urlParams.get('IDEmisorFactura');
                    if (urlParams.get('NumSerieFactura')) state.data.numFactura = urlParams.get('NumSerieFactura');
                    if (urlParams.get('FechaExpedicionFactura')) {
                        // Formato suele ser DD-MM-YYYY, convertimos a YYYY-MM-DD para el input
                        const parts = urlParams.get('FechaExpedicionFactura').split('-');
                        if(parts.length===3) state.data.fechaEmision = `${parts[2]}-${parts[1]}-${parts[0]}`;
                    }
                    if (urlParams.get('ImporteTotal')) {
                        state.data.total = parseFloat(urlParams.get('ImporteTotal'));
                        // Deducción inversa básica para rellenar (asumiendo 21% sin IRPF)
                        state.data.base = +(state.data.total / 1.21).toFixed(2);
                        // Limpiamos items y ponemos uno genérico para que cuadre
                        state.data.items = [{ desc: 'Conceptos según registro electrónico (Extraído de QR)', qty: 1, price: state.data.base }];
                    }
                } else {
                    // Fallback si es texto plano o formato desconocido pero validamos que contenga "ES"
                    console.log("Formato QR no estándar, se usa como imagen.");
                }

                // Actualizar UI
                document.getElementById('qr-status-badge').classList.remove('hidden');
                updateFormUI();
                renderFormItems();
                calculateTotals();
                lockQRFields();

            } catch (error) {
                console.error("Error parseando QR:", error);
                showToast('Error leyendo datos específicos del QR.', 'error');
            }
        }

        function updateItem(index, field, value) {
            state.data.items[index][field] = field === 'desc' ? value : parseFloat(value) || 0;
            calculateTotals();
        }

            state.data.items.push({ desc: '', qty: 1, price: 0 });

        function removeItem(index) {
            state.data.items.splice(index, 1);
            renderFormItems();
            calculateTotals();
        }

        function calculateTotals() {
            // Actualizar state vars desde DOM por si acaso
            state.data.ivaPct = parseInt(document.getElementById('ivaPct').value) || 0;
            state.data.irpfPct = parseInt(document.getElementById('irpfPct').value) || 0;

            let base = 0;
            state.data.items.forEach(item => {
                base += (item.qty * item.price);
            });
            
            state.data.base = base;
            state.data.ivaCuota = base * (state.data.ivaPct / 100);
            state.data.irpfCuota = base * (state.data.irpfPct / 100);
            state.data.total = base + state.data.ivaCuota - state.data.irpfCuota;

            // Update UI totals
            document.getElementById('baseTotal').value = state.data.base.toFixed(2);
            document.getElementById('totalDisplay').textContent = state.data.total.toFixed(2) + ' €';

            updatePreview();
        }

        // --- EXPORTACIÓN A PDF ---
        function exportPDF() {
            const element = document.getElementById('invoice-preview');
            
            // Notificación visual de inicio
            showToast('Preparando PDF legal...', 'info');
            
            // Configuración html2pdf optimizada
            const opt = {
                margin:       0,
                filename:     `Factura_${state.data.numFactura || 'Legal'}.pdf`,
                image:        { type: 'jpeg', quality: 0.98 },
                html2canvas:  { scale: 2, useCORS: true, letterRendering: true },
                jsPDF:        { unit: 'mm', format: 'a4', orientation: 'portrait' }
            };

            // Asegurar que si hay template oscuro (Tech), se renderiza bien en el PDF
            // html2pdf captura lo que ve. El A4 container está aislado de Tailwind dark mode.
            
            html2pdf().set(opt).from(element).save().then(() => {
                showToast('PDF exportado con éxito.', 'success');
            }).catch(err => {
                console.error("Error al exportar PDF:", err);
                showToast('Error al exportar. Intente de nuevo.', 'error');
            });
        }

        // --- COPIADO DE DATOS (Para Gestoría) ---
        function copyPlainData() {
            const d = state.data;
            let text = `FACTURA: ${d.numFactura}\nFECHA: ${formatDate(d.fechaEmision)}\n`;
            text += `EMISOR: ${d.nombreEmisor} (${d.nifEmisor})\n`;
            text += `CLIENTE: ${d.nombreCliente} (${d.nifCliente})\n\n`;
            text += `BASE IMPONIBLE: ${d.base.toFixed(2)} €\n`;
            text += `IVA (${d.ivaPct}%): ${d.ivaCuota.toFixed(2)} €\n`;
            if (d.irpfPct > 0) text += `IRPF (${d.irpfPct}%): -${d.irpfCuota.toFixed(2)} €\n`;
            text += `TOTAL: ${d.total.toFixed(2)} €\n\n`;
            text += `Generado por VeriFactu Design.`;

            navigator.clipboard.writeText(text).then(() => {
                showToast('Datos copiados al portapapeles.', 'success');
            }).catch(err => {
                // Fallback para iFrames
                const textArea = document.createElement("textarea");
                textArea.value = text;
                document.body.appendChild(textArea);
                textArea.select();
                try {
                    document.execCommand('copy');
                    showToast('Datos copiados al portapapeles.', 'success');
                } catch (err) {
                    showToast('Error al copiar.', 'error');
                }
                document.body.removeChild(textArea);
            });
        }
